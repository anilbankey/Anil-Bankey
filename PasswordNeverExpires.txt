Import-Module ActiveDirectory

# Output file
$OutputFile = "C:\Temp\PasswordNeverExpires_Users.csv"

# Get users
$Users = Get-ADUser -Filter {PasswordNeverExpires -eq $true} -Properties `
    Name,
    SamAccountName,
    Enabled,
    PasswordNeverExpires,
    PasswordLastSet,
    LastLogonDate,
    Description

# Prepare output
$Results = foreach ($User in $Users) {

    # Calculate Password Age
    $PasswordAge = $null
    if ($User.PasswordLastSet) {
        $PasswordAge = (New-TimeSpan -Start $User.PasswordLastSet -End (Get-Date)).Days
    }

    [PSCustomObject]@{
        UserName            = $User.SamAccountName
        DisplayName         = $User.Name
        Enabled             = $User.Enabled
        PasswordNeverExpire = $User.PasswordNeverExpires
        PasswordLastSet     = $User.PasswordLastSet
        PasswordAgeDays     = $PasswordAge
        LastLogonDate       = $User.LastLogonDate
        Description         = $User.Description
    }
}

# Display
$Results | Format-Table -AutoSize

# Export
$Results | Export-Csv $OutputFile -NoTypeInformation -Encoding UTF8

Write-Host "`nReport generated at: $OutputFile" -ForegroundColor Green
